# Web MLE — Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Build the MLE / Lista Liberal campaign website: a single-page Flask app (Jinja + Tailwind) with a public contact form and a private admin panel, deployed to Vercel as a Python serverless function with Postgres.

**Architecture:** Flask application factory. Jinja renders the landing from a Python content module. Backend persists contact submissions and serves an authenticated admin listing. Static assets served by Vercel CDN; everything else runs in one serverless function. Rate limiting is DB-backed (serverless-safe).

**Tech Stack:** Flask 3, SQLAlchemy 2, Flask-Login, Flask-WTF, Flask-Migrate, Jinja2, Tailwind CSS v4 (CLI build), vanilla JS, pytest, Vercel + Vercel Postgres (Neon).

**Spec:** `docs/superpowers/specs/2026-10-13-mle-web-design.md`

## Global Constraints

- Python version floor: `>=3.11`.
- Dependency floors: Flask `>=3.0`, SQLAlchemy `>=2.0`, Flask-SQLAlchemy `>=3.1`, Flask-Login `>=0.6`, Flask-WTF `>=1.2`, Flask-Migrate `>=4.0`, python-dotenv `>=1.0`, psycopg `[binary]>=3.1`, gunicorn `>=21`, pytest `>=8`.
- Node dev dependency: `tailwindcss@^4`, `@tailwindcss/cli@^4`.
- Brand colors (exact): `--color-violeta: #7C3AED`, `--color-violeta-oscuro: #5B21B6`, `--color-violeta-claro: #A78BFA`, `--color-negro: #0B0B0F`, `--color-blanco: #FFFFFF`.
- Lema (exact): «Nada de promesas, puros resultados».
- Section anchor ids (exact, in order): `#inicio`, `#nosotros`, `#propuestas`, `#equipo`, `#plan`, `#transparencia`, `#sumate`, plus footer.
- Submission `tipo` values (exact): `sugerencia`, `reclamo`, `idea`.
- Rate limits (exact): `contact` = 5 per IP per 15 min; `login` = 10 per IP per 15 min.
- No external CDN assets (CSP `default-src 'self'`). Fonts must be system stack or self-hosted.
- Secrets only via environment variables. `SECRET_KEY` required in prod (app raises if missing).
- All user input `strip()`ed; no user HTML rendered (Jinja autoescape).
- Commits use Conventional Commits.

```python
from flask import Flask
from sqlalchemy.pool import StaticPool

from app.config import CONFIG_MAP
from app.extensions import csrf, db, login_manager, migrate


def create_app(config_name: str = "dev") -> Flask:
    app = Flask(__name__)
    app.config.from_object(CONFIG_MAP[config_name])

    if config_name == "prod":
        CONFIG_MAP["prod"]()  # validate env

    # SQLite in-memory needs a single shared connection for tests
    if app.config["SQLALCHEMY_DATABASE_URI"] == "sqlite://":
        app.config["SQLALCHEMY_ENGINE_OPTIONS"] = {
            "poolclass": StaticPool,
            "connect_args": {"check_same_thread": False},
        }
    else:
        app.config.setdefault(
            "SQLALCHEMY_ENGINE_OPTIONS",
            {"pool_pre_ping": True, "poolclass": StaticPool}
            if app.config["SQLALCHEMY_DATABASE_URI"].startswith("sqlite")
            else {"pool_pre_ping": True},
        )

    db.init_app(app)
    login_manager.init_app(app)
    csrf.init_app(app)
    migrate.init_app(app, db)

    from app.models import Submission, RateLimitHit  # noqa: F401

    from app.routes.admin import admin_bp
    from app.routes.contact import contact_bp
    from app.routes.main import main_bp

    app.register_blueprint(main_bp)
    app.register_blueprint(contact_bp)
    app.register_blueprint(admin_bp)

    return app
```

from datetime import datetime, timedelta, timezone

from app.extensions import db
from app.models import RateLimitHit


def _now():
    return datetime.now(timezone.utc)


def _as_aware(dt):
    return dt if dt.tzinfo else dt.replace(tzinfo=timezone.utc)


def check_rate_limit(ip: str, scope: str, limit: int, window_seconds: int) -> bool:
    since = _now() - timedelta(seconds=window_seconds)
    count = (
        RateLimitHit.query.filter(
            RateLimitHit.ip == ip,
            RateLimitHit.scope == scope,
        )
        .all()
    )
    recent = [h for h in count if _as_aware(h.created_at) >= since]
    if len(recent) >= limit:
        return False
    db.session.add(RateLimitHit(ip=ip, scope=scope))
    db.session.commit()
    return True
```

import smtplib
from email.message import EmailMessage

from flask import current_app


def send_submission_notification(submission) -> bool:
    app = current_app
    if app.config.get("MAIL_SUPPRESS_SEND"):
        return True
    host = app.config.get("SMTP_HOST")
    mail_to = app.config.get("MAIL_TO")
    if not host or not mail_to:
        return False

    msg = EmailMessage()
    msg["Subject"] = f"Nuevo mensaje MLE — {submission.tipo}"
    msg["From"] = app.config.get("SMTP_USER") or mail_to
    msg["To"] = mail_to
    msg.set_content(
        "Nuevo mensaje recibido en la web del MLE.\n\n"
        f"Nombre: {submission.nombre}\n"
        f"Contacto: {submission.contacto}\n"
        f"Tipo: {submission.tipo}\n\n"
        f"Mensaje:\n{submission.mensaje}\n"
    )

    try:
        with smtplib.SMTP(host, app.config.get("SMTP_PORT", 587), timeout=10) as smtp:
            smtp.starttls()
            user = app.config.get("SMTP_USER")
            password = app.config.get("SMTP_PASS")
            if user and password:
                smtp.login(user, password)
            smtp.send_message(msg)
        return True
    except Exception:  # best-effort: never break the request
        app.logger.exception("Fallo al enviar notificación de envío")
        return False
```

from flask import current_app
from flask_login import UserMixin
from werkzeug.security import check_password_hash

from app.extensions import login_manager


class AdminUser(UserMixin):
    def __init__(self, username):
        self.id = username


@login_manager.user_loader
def load_user(user_id):
    if user_id == current_app.config.get("ADMIN_USER"):
        return AdminUser(user_id)
    return None


def authenticate(username: str, password: str):
    cfg_user = current_app.config.get("ADMIN_USER")
    cfg_hash = current_app.config.get("ADMIN_PASSWORD_HASH")
    if not username or not cfg_user or username != cfg_user:
        return None
    if not cfg_hash or not check_password_hash(cfg_hash, password):
        return None
    return AdminUser(username)
